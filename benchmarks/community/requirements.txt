# Python dependencies for the Clio Coder community benchmark adapters.
#
# The adapters are run as scripts, not installed as a package. Prefer the
# README-documented `uv run --no-project [--with ...] python ...` commands so
# benchmark invocations do not depend on a machine-specific interpreter. This
# file remains a compatibility list for operators who still want a prebuilt venv.
# Deps are grouped by adapter; you only need the group for the track you run.

# --- swe-bench-lite (swebench_clio.py, recompute_patches.py) ---
datasets            # load princeton-nlp/SWE-bench_Lite
swebench            # harness for `python -m swebench.harness.run_evaluation`

# --- terminal-bench (tb_clio_agent/) ---
terminal-bench      # provides the `tb` CLI and AbstractInstalledAgent base

# --- scicode (scicode_clio.py grade path) ---
# Only needed to grade against the official HDF5 targets; task generation and
# the JSON-manifest smoke path need none of these.
h5py                # process_hdf5_to_tuple over the official test_data.h5
scipy               # SciCode reference numerics
numpy               # SciCode reference numerics

# --- human-eval (humaneval_clio.py official evaluator path) ---
human-eval @ git+https://github.com/openai/human-eval.git
